cmake_minimum_required(VERSION 3.22)
project(sptflite CXX)

set(TFLITE_LIB ${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/arm64-v8a/libtensorflowlite_jni.so)
set(QNN_DELEGATE_LIB ${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/arm64-v8a/libQnnTFLiteDelegate.so)
add_library(sptflite SHARED sp_tflite.cpp)
target_include_directories(sptflite PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tflite_headers ${CMAKE_CURRENT_SOURCE_DIR}/qnn_tflite)
target_compile_options(sptflite PRIVATE -O3 -ffast-math -std=c++17)
target_link_libraries(sptflite ${TFLITE_LIB} ${QNN_DELEGATE_LIB} log android dl)
set(CMAKE_CXX_STANDARD 17)

# ---------------- OpenVINS VIO (SuperPoint front-end + MSCKF backend) ----------------
set(OV ${CMAKE_CURRENT_SOURCE_DIR}/openvins)
file(GLOB_RECURSE OV_CORE_SRC ${OV}/ov_core/src/*.cpp)
list(FILTER OV_CORE_SRC EXCLUDE REGEX "dummy\\.cpp$")
list(FILTER OV_CORE_SRC EXCLUDE REGEX "test_")
list(FILTER OV_CORE_SRC EXCLUDE REGEX "sim/BsplineSE3\\.cpp$")
set(OV_INIT_SRC ${OV}/ov_init/src/init/InertialInitializer.cpp ${OV}/ov_init/src/static/StaticInitializer.cpp)
file(GLOB_RECURSE OV_MSCKF_SRC ${OV}/ov_msckf/src/*.cpp)
list(FILTER OV_MSCKF_SRC EXCLUDE REGEX "run_subscribe_msckf\\.cpp$")
list(FILTER OV_MSCKF_SRC EXCLUDE REGEX "dummy\\.cpp$")
list(FILTER OV_MSCKF_SRC EXCLUDE REGEX "/ros/")
list(FILTER OV_MSCKF_SRC EXCLUDE REGEX "ros[0-9_]")


list(FILTER OV_CORE_SRC EXCLUDE REGEX "/sim/")
list(FILTER OV_INIT_SRC EXCLUDE REGEX "/sim/")
list(FILTER OV_MSCKF_SRC EXCLUDE REGEX "/sim/")

list(FILTER OV_INIT_SRC EXCLUDE REGEX "test_")
list(FILTER OV_MSCKF_SRC EXCLUDE REGEX "test_")
list(FILTER OV_MSCKF_SRC EXCLUDE REGEX "run_")
list(FILTER OV_CORE_SRC EXCLUDE REGEX "test_")
add_library(openvins_jni SHARED ${OV_CORE_SRC} ${OV_INIT_SRC} ${OV_MSCKF_SRC} ${CMAKE_CURRENT_SOURCE_DIR}/superpoint_track.cpp ${CMAKE_CURRENT_SOURCE_DIR}/vio_jni.cpp)
set_target_properties(openvins_jni PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_compile_definitions(openvins_jni PRIVATE __ANDROID__)
target_compile_options(openvins_jni PRIVATE -O3 -fexceptions -frtti -Wno-unused-parameter)
target_include_directories(openvins_jni PRIVATE ${OV}/ov_core/src ${OV}/ov_init/src ${OV}/ov_msckf/src ${CMAKE_CURRENT_SOURCE_DIR} ${OV}/thirdparty ${OV}/thirdparty/opencv_include ${OV}/thirdparty/boost_include)

add_library(lib_opencv SHARED IMPORTED)
set_target_properties(lib_opencv PROPERTIES IMPORTED_LOCATION ${OV}/thirdparty/opencv_lib/arm64-v8a/libopencv_vio4.so)
add_library(boost_system SHARED IMPORTED)
set_target_properties(boost_system PROPERTIES IMPORTED_LOCATION ${OV}/thirdparty/boost_lib/arm64-v8a/libboost_system.so)
add_library(boost_filesystem SHARED IMPORTED)
set_target_properties(boost_filesystem PROPERTIES IMPORTED_LOCATION ${OV}/thirdparty/boost_lib/arm64-v8a/libboost_filesystem.so)
add_library(boost_thread SHARED IMPORTED)
set_target_properties(boost_thread PROPERTIES IMPORTED_LOCATION ${OV}/thirdparty/boost_lib/arm64-v8a/libboost_thread.so)
add_library(boost_chrono SHARED IMPORTED)
set_target_properties(boost_chrono PROPERTIES IMPORTED_LOCATION ${OV}/thirdparty/boost_lib/arm64-v8a/libboost_chrono.so)
add_library(boost_date_time SHARED IMPORTED)
set_target_properties(boost_date_time PROPERTIES IMPORTED_LOCATION ${OV}/thirdparty/boost_lib/arm64-v8a/libboost_date_time.so)
add_library(boost_atomic SHARED IMPORTED)
set_target_properties(boost_atomic PROPERTIES IMPORTED_LOCATION ${OV}/thirdparty/boost_lib/arm64-v8a/libboost_atomic.so)

target_link_libraries(openvins_jni lib_opencv boost_system boost_filesystem boost_thread boost_chrono boost_date_time boost_atomic log android dl)
